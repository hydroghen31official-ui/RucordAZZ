function ProductCard({ product, t, lang, isFavorite, onToggleFav, onAddToCart, onOpenDetails }) {
  const finalPrice = product.discountPrice || product.price;

  return (
    <div className="group rounded-3xl bg-zinc-900/40 border border-zinc-800/80 hover:border-rose-500/50 hover:bg-zinc-900 transition-all duration-300 overflow-hidden flex flex-col justify-between shadow-lg">
      <div className="relative overflow-hidden aspect-[4/3] bg-zinc-950">
        <img
          src={product.image}
          alt={product.title}
          className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
        />

        <div className="absolute top-3 left-3 flex flex-col gap-1.5">
          {product.discountPrice && (
            <span className="bg-rose-600 text-white text-[10px] font-black px-2 py-0.5 rounded-md shadow-md">
              -{Math.round(((product.price - product.discountPrice) / product.price) * 100)}%
            </span>
          )}
          <span className="bg-zinc-950/80 backdrop-blur-md text-emerald-400 text-[9px] font-extrabold px-2 py-0.5 rounded-md border border-zinc-800">
            {t.product.fastDelivery}
          </span>
        </div>

        <button
          onClick={(e) => {
            e.stopPropagation();
            onToggleFav(product.id);
          }}
          className={`absolute top-3 right-3 w-8 h-8 rounded-full flex items-center justify-center transition-all ${
            isFavorite
              ? 'bg-rose-600 text-white shadow-lg'
              : 'bg-zinc-950/70 text-zinc-400 hover:text-white'
          }`}
        >
          <Heart className={`w-4 h-4 ${isFavorite ? 'fill-white' : ''}`} />
        </button>
      </div>

      <div className="p-4 flex-1 flex flex-col justify-between space-y-3">
        <div>
          <h4 
            onClick={() => onOpenDetails(product)}
            className="text-sm font-bold text-white hover:text-rose-400 transition-colors line-clamp-1 cursor-pointer"
          >
            {product.title}
          </h4>
          <p className="text-[11px] text-zinc-400 line-clamp-2 mt-1 leading-relaxed">
            {lang === 'az' ? product.descAz : product.descTr}
          </p>
        </div>

        <div className="pt-2 border-t border-zinc-850 flex items-center justify-between">
          <div>
            <div className="flex items-baseline gap-1.5">
              <span className="text-base font-black text-white">{finalPrice.toFixed(2)}</span>
              <span className="text-xs font-bold text-rose-500">AZN</span>
            </div>
            {product.discountPrice && (
              <span className="text-[10px] text-zinc-500 line-through">
                {product.price.toFixed(2)} AZN
              </span>
            )}
          </div>

          <button
            onClick={() => onAddToCart(product)}
            className="p-2.5 rounded-xl bg-zinc-800 hover:bg-rose-600 text-zinc-200 hover:text-white transition-all shadow-md group-hover:bg-rose-600 group-hover:text-white"
            title={t.product.addToCart}
          >
            <ShoppingBag className="w-4 h-4" />
          </button>
        </div>
      </div>
    </div>
  );
}
